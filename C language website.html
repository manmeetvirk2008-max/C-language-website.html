Q1. Convert a decimal number into binary.

#include <stdio.h>

int main() {
    int n, binary=0, place=1;

    printf("Enter a decimal number: ");
    scanf("%d", &n);

    while(n>0) {
        binary += (n%2)*place;
        n /= 2;
        place *= 10;
    }

    printf("Binary = %d", binary);

    return 0;
}

Input: 10

Output: Binary = 1010

Q2. Find the prime divisors of a number.

#include <stdio.h>

int main() {
    int n, i;

    printf("Enter a number: ");
    scanf("%d", &n);

    printf("Prime divisors: ");

    for(i=2; i<=n; i++) {
        if(n%i==0) {
            int j, prime=1;

            for(j=2; j*j<=i; j++) {
                if(i%j==0) {
                    prime=0;
                    break;
                }
            }

            if(prime)
                printf("%d ", i);
        }
    }

    return 0;
}

Input: 12

Output: Prime divisors: 2 3

        {
            flag = 0;
            break;
        }
    }

    if(flag)
        printf("Palindrome");
    else
        printf("Not a palindrome");

    return 0;
}
